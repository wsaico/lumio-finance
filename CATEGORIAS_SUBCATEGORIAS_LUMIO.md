## 🔧 IMPLEMENTACIÓN EN BASE DE DATOS

### Schema SQL

```sql
-- Categorías principales
CREATE TABLE expense_categories (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id),
  name VARCHAR(100) NOT NULL,
  icon VARCHAR(50) NOT NULL,
  color VARCHAR(7) NOT NULL,
  is_system BOOLEAN DEFAULT true,
  sort_order INTEGER,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Subcategorías
CREATE TABLE subcategories (
  id UUID PRIMARY KEY,
  category_id UUID REFERENCES expense_categories(id),
  name VARCHAR(100) NOT NULL,
  is_system BOOLEAN DEFAULT true,
  sort_order INTEGER,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Seed data example
INSERT INTO expense_categories (name, icon, color, is_system, sort_order) VALUES
  ('Alimentación y Bebidas', 'utensils', '#EF4444', true, 1),
  ('Vivienda', 'home', '#F59E0B', true, 2),
  ('Transporte', 'car', '#3B82F6', true, 3);

INSERT INTO subcategories (category_id, name, is_system, sort_order) VALUES
  -- Para Alimentación
  ('[category_id_alimentacion]', 'Supermercado', true, 1),
  ('[category_id_alimentacion]', 'Bodega', true, 2),
  ('[category_id_alimentacion]', 'Restaurant', true, 3);
```
